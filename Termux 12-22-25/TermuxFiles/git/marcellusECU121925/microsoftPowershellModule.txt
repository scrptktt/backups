function Enable-ProcessCreationEvent
{
 $Query =
<event-class>                               <__InstanceCreationEvent>>  https://learn.microsoft.com/en-us/windows/win32/wmisdk/--instancecreationevent
 New-Object System.Management.WqlEventQuery "__InstanceCreationEvent",
 <event-class>
 (New-Object TimeSpan 0,0,1),
 "TargetInstance isa 'Win32_Process'"
                     <<Win32_Process>   https://learn.microsoft.com/en-us/windows/win32/cimwin32prov/win32-process
                <isa>                   https://learn.microsoft.com/en-us/windows/win32/wmisdk/isa-operator-for-schema-queries

 $ProcessWatcher =
 New-Object System.Management.ManagementEventWatcher $Query
 
 $Identifier = "WMI.ProcessCreated"
 
 Register-ObjectEvent
 $ProcessWatcher "EventArrived" -SupportEvent $Identifier `
 -Action
 {
  [void] (New-Event -SourceId "PowerShell.ProcessCreated" -Sender $args[0] `
  -EventArguments $args[1].SourceEventArgs.NewEvent.TargetInstance)
 }
}
